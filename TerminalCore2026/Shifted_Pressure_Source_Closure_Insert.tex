\subsubsection*{The axis primitive and the two remaining direct linear pressure sources}
\label{subsec:direct-pressure-shifted}
The shifted pressure space must control the other terms of the actual
pressure equation, not only delayed viscosity. We now estimate the
swirl-square primitive and both direct linear pressure sources, without
a radial Taylor cutoff. All norms are the induced norms defined in
\eqref{eq:norm} and \eqref{eq:shifted-P-norm}.

\begin{lemma}[Sharp shifted-norm axis primitive]
\label{lem:shifted-axis-primitive}
For $R,\rho>0$ and $\mathcal IF=\int_0^X F(x,\eta)\,dx$,
\[
\|\mathcal IF\|_{\mathcal P}
\le\frac{16}{3}R\rho\,\|F\|_{\mathcal B}.
\]
The coefficient constant $16/3$ is sharp, attained for $F=1$.
The map raises radial valuation by one.
\end{lemma}
\begin{proof}
The exact coefficient ratio is
\[
\frac{w_{a,b}}{(a+1)w_{a+1,b+1}}
=R\rho\frac{(a+2)^2}{(a+1)^3}
\frac{(b+2)^2}{(b+1)^2}
\frac{(2a+1)(2a+2)}
{(2a+b+1)(2a+b+2)(2a+b+3)}.
\]
For fixed $a$ this decreases with $b$. At $b=0$ it is
$4R\rho(a+2)^2/[(a+1)^3(2a+3)]$.
The inequality $3(a+2)^2\le4(a+1)^3(2a+3)$ follows
from the exact nonnegative factorization
\[
4(a+1)^3(2a+3)-3(a+2)^2
=a(8a^3+36a^2+57a+32).
\]
At $a=b=0$ the ratio equals $16R\rho/3$.
\end{proof}

Applying the weighted multiplication theorem \eqref{eq:algebra64}
to the exact swirl-square source in (27) yields
\begin{equation}\label{eq:swirl-P-source}
\|\Pi_n^{\mathrm{sw}}\|_{\mathcal P}
\le\frac{1024}{3}R\rho|C|^{-2}
\sum_{i+j=n}\|\varphi_i\|_{\mathcal B}
\|\varphi_j\|_{\mathcal B},
\qquad
\Pi_n^{\mathrm{sw}}=C^{-2}\int_0^X\sum_{i+j=n}\varphi_i\varphi_j\,dx.
\end{equation}
The sum includes both current/leading ordered pairs and all
strictly positive pairs. This statement is conditional on finite
norms of the coefficients it contains.

\begin{proposition}[Exact radial-Laplacian pressure cancellation]
\label{prop:lap-pressure-shifted}
For $k\ge1$, the pressure primitive contributed by
$-2XV_{k,XX}$ to $\Omega_k$ is precisely
$\Pi_{k+1}^{\mathrm{lap}}=\partial_XV_k$.
For $0<h<1/100$ and $0<\rho\le1/4$, the complete composite maps
$I_k(\mathcal B)$ into $I_k(\mathcal P)$ and
\begin{equation}\label{eq:lap-pressure-shifted-bound}
\|\Pi_{k+1}^{\mathrm{lap}}\|_{\mathcal P}
\le\frac{80}{27}(32\rho+13)\|U_k\|_{\mathcal B}.
\end{equation}
\end{proposition}
\begin{proof}
In (27), the isolated term gives
$\Pi'_{k+1}=V_{k,XX}$. Since $V_k=O(X^{k+1})$
and $k\ge1$, the axis constant vanishes, proving the
identity. At input radial degree $a\ge k$, put
$r=a+1$, $\lambda_k=2hk$ and $F_a=(U_k)_a$.
The exact radial-velocity formula (24) gives
\[
(\Pi_{k+1}^{\mathrm{lap}})_a
=(r-D-\lambda_k)BF_a-G\partial_\eta F_a,
\quad B=\frac{2\eta}{L},\quad G=\frac{1-\eta^2}{L}.
\]
Here $0\le r-D-\lambda_k\le r$. The shift absorbs
one axial derivative exactly and the radial factor obeys
\[
\frac{r w_{a,b}}{w_{a,b+1}}
=\rho\frac{r(b+2)^2}{(b+1)^2(2a+b+1)}
\le4\rho.
\]
On $|\eta|\le3$ one has $|B|<8$, $|G|<13$.
Their derivatives on $[-1,1]$ are bounded respectively
by $8\ell!$, $13\ell!$. Apply the positive Leibniz
multiplier estimate \eqref{eq:P-multiplier}, also to the
radial-factor weight comparison. Its sum is
$(1+\rho)/(1-\rho)^3\le80/27$, proving the claim.
\end{proof}

\begin{theorem}[Uniform direct time-pressure source]
\label{thm:time-pressure-shifted}
Under the same parameter hypotheses and for every $k\ge1$,
the pressure primitive from $T_{0,k}V_k$ in (28) belongs
to $I_{k+1}(\mathcal P)$ and satisfies
\begin{equation}\label{eq:time-pressure-shifted-bound}
\|\Pi_{k+1}^{\mathrm{time}}\|_{\mathcal P}
\le\frac{1280R}{27}
\left(\frac{32}{3}\rho+12+\frac8\rho\right)
\|U_k\|_{\mathcal B}.
\end{equation}
\end{theorem}
\begin{proof}
For input degree $a\ge k$ set $r=a+1$,
$\alpha=1-(D+\lambda_k)/r$ and $H=D\eta/L$.
Writing $W_a=\alpha BF_a-(G/r)F_a'$,
the exact operator at radial degree $r$ is
$T_{0,k}^{(r)}=(r-\lambda_k)/L+H\partial_\eta$.
One axis primitive in (27) gives
\begin{equation}\label{eq:time-pressure-exact-normal}
(\Pi_{k+1}^{\mathrm{time}})_{a+1}
=-\frac1{2r}(c_0F_a+c_1F_a'+c_2F_a''),
\end{equation}
with all variable-coefficient derivatives retained:
\begin{align*}
c_0&=\frac{r-\lambda_k}{L}\alpha B+H\alpha B',\\
c_1&=-\frac{r-\lambda_k}{rL}G
       +H(\alpha B-G'/r),\\
c_2&=-HG/r.
\end{align*}
The source indices satisfy
$0\le\alpha\le1$, $0\le r-\lambda_k\le r$.
Cancellation of the weight factorials gives
\begin{equation}\label{eq:time-pressure-three-ratios}
\frac{w_{a,b}}{w_{a+1,b+1}}
\le\frac{16}{3}R\rho,\qquad
\frac{w_{a,b+1}}{r w_{a+1,b+1}}\le4R,\qquad
\frac{w_{a,b+2}}{r^2w_{a+1,b+1}}\le\frac{8R}{\rho}.
\end{equation}
For the first inequality the $b=0$ radial polynomial difference
is $a(8a^2+25a+20)\ge0$. For the second,
$(a+2)^2/r^2\le4$ and the two denominator factorial
terms dominate $(2a+1)(2a+2)$. For the third use
$(2a+1)(2a+2)/(2a+b+3)\le2r$.
On $|\eta|\le3$ each of the six rational multipliers
$B/L$, $HB'$, $G/L$, $HB$, $HG'$, $HG$
is bounded by $32$: use $|L|>41/50$, $|H|<2$,
$|B'|<4$ and $|G'|<10$.
Cauchy bounds on $[-1,1]$ are then
$32\ell!$ for each. The normal form has two
zeroth-jet, three first-jet and one second-jet terms,
carrying respectively radial factors at most $1$,
$1/r$ and $1/r^2$, with outer prefactor $1/2$.
Apply the three estimates above and the unsigned
Leibniz sum $80/27$ to obtain the displayed constant.
\end{proof}

\paragraph{Pressure-source ledger after these estimates.}
Together with the previous bound
\eqref{eq:shifted-delayed-bound}, these results control
all three direct linear sources of $\Omega_{n-1}$
in the shifted pressure norm, for $n\ge3$:
\begin{equation}\label{eq:direct-pressure-P-bound}
\|\Pi_n^{\mathrm{direct}}\|_{\mathcal P}
\le(C_{\mathrm{P,time}}+C_{\mathrm{P,lap}})
\|U_{n-1}\|_{\mathcal B}
+C_{\mathrm{P,del}}\|U_{n-2}\|_{\mathcal B},
\quad C_{\mathrm{P,lap}}=\frac{80}{27}(32\rho+13).
\end{equation}
The full swirl-square contribution is bounded by
\eqref{eq:swirl-P-source}. Still unproved in the shifted
pressure norm are the bare primitives of the five leading-profile
interactions and the positive-order quadratic terms in
$\Omega_{n-1}$. Their earlier bounds after the final
$J_1Z$ regularization do not themselves establish these
pressure-norm estimates. The complete two-space recurrence,
imported profile radii, and physical-coordinate pressure
reconstruction therefore remain open.
